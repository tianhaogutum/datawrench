#!/usr/bin/env python3
"""Replay this bundle locally. Exit 0: same target; 1: absent; 2: invalid/error."""
import argparse
import hashlib
import importlib.metadata
import json
import math
import os
from pathlib import Path
import re
import signal
import subprocess
import sys


def sha256(path):
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def check_manifest(manifest):
    """Reject malformed metadata before executing copied validation rules."""
    try:
        if type(manifest["format_version"]) is not int or manifest["format_version"] != 1:
            raise ValueError("unsupported format version")
        schema = manifest["schema_file"]
        if schema not in {"schema.py", "schema.yaml"}:
            raise ValueError("invalid schema member")
        required = {"data.csv", schema, "read-options.json", "reduction.jsonl",
                    "_validator.py", "replay.py", "requirements.txt"}
        if not isinstance(manifest["files"], dict) or set(manifest["files"]) != required:
            raise ValueError("required member hashes are missing or unexpected")
        for digest in manifest["files"].values():
            if not isinstance(digest, str) or not re.fullmatch(r"[0-9a-f]{64}", digest):
                raise ValueError("invalid member hash")
        versions = manifest["validator_versions"]
        if not isinstance(versions, dict) or not {"pandas", "pandera", "numpy"} <= set(versions):
            raise ValueError("required validator versions are missing")
        for name, version in versions.items():
            if not re.fullmatch(r"[A-Za-z0-9][A-Za-z0-9_.-]*", name):
                raise ValueError("invalid dependency name")
            if not isinstance(version, str) or not 0 < len(version) <= 100:
                raise ValueError("invalid dependency version")
        target = manifest["target"]["id"]
        if not isinstance(target, str) or not re.fullmatch(r"pa:[0-9a-f]{64}", target):
            raise ValueError("invalid target identity")
        description = manifest["target"]
        required_fields = {"id", "schema_context", "column", "check", "check_index", "reason"}
        if not required_fields <= set(description) <= required_fields | {"unique_columns"}:
            raise ValueError("missing or unknown target description fields")
        for field in ("schema_context", "reason"):
            if not isinstance(description[field], str) or not description[field]:
                raise ValueError("invalid target description")
        for field in ("column", "check"):
            if description[field] is not None and not isinstance(description[field], str):
                raise ValueError("invalid target description")
        if description["check_index"] is not None and type(description["check_index"]) is not int:
            raise ValueError("invalid target check index")
        if "unique_columns" in description:
            columns = description["unique_columns"]
            if not isinstance(columns, list) or not all(isinstance(c, str) for c in columns):
                raise ValueError("invalid uniqueness columns")
        identity = {key: value for key, value in manifest["target"].items() if key != "id"}
        encoded = json.dumps(identity, sort_keys=True, separators=(",", ":")).encode()
        if target != "pa:" + hashlib.sha256(encoded).hexdigest():
            raise ValueError("target identity does not match its description")
        repeats = manifest["reduction"]["stability_runs"]
        timeout = manifest["reduction"]["timeout_seconds"]
        if type(repeats) is not int or not 2 <= repeats <= 10:
            raise ValueError("stability runs must be between 2 and 10")
        if type(timeout) not in {int, float} or not 0 < timeout <= 300 or not math.isfinite(timeout):
            raise ValueError("timeout must be finite and between 0 and 300 seconds")
        source_rows, rows = manifest["source"]["rows"], manifest["result"]["rows"]
        if type(source_rows) is not int or type(rows) is not int or not 0 <= rows <= source_rows:
            raise ValueError("invalid row counts")
        previous = total = 0
        intervals = manifest["result"]["source_record_ranges"]
        if not isinstance(intervals, list):
            raise ValueError("record ranges must be a list")
        for interval in intervals:
            if not isinstance(interval, list) or len(interval) != 2:
                raise ValueError("invalid record range")
            start, end = interval
            if type(start) is not int or type(end) is not int or not previous < start <= end <= source_rows:
                raise ValueError("invalid record range bounds")
            previous = end
            total += end - start + 1
        if total != rows:
            raise ValueError("row count does not match source record ranges")
        python = manifest["environment"]["python"]
        if not isinstance(python, str) or not re.fullmatch(r"\d+\.\d+\.\d+", python):
            raise ValueError("invalid Python version")
    except (KeyError, TypeError, ValueError) as exc:
        raise ValueError("invalid bundle manifest: " + str(exc)) from exc


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--allow-version-drift", action="store_true")
    parser.add_argument("--bundle", type=Path, default=Path(__file__).resolve().parent,
                        help="bundle directory (defaults to this script's directory)")
    args = parser.parse_args()
    root = args.bundle.resolve()
    try:
        manifest_path = root / "manifest.json"
        if manifest_path.stat().st_size > 16 * 1024 * 1024:
            raise ValueError("bundle manifest exceeds the 16 MiB limit")
        manifest = json.loads(manifest_path.read_text())
        check_manifest(manifest)
        for name, digest in manifest["files"].items():
            if Path(name).name != name or (root / name).is_symlink() or not (root / name).is_file():
                raise ValueError("invalid bundle member: " + name)
            if sha256(root / name) != digest:
                raise ValueError("bundle member changed: " + name)
        if not args.allow_version_drift:
            expected_python = manifest["environment"]["python"].split(".")[:2]
            if expected_python != [str(v) for v in sys.version_info[:2]]:
                raise ValueError("Python minor version differs from the recorded environment")
            for name, expected in manifest["validator_versions"].items():
                installed = importlib.metadata.version(name)
                if installed != expected:
                    raise ValueError(f"{name} version {installed}; expected {expected}")
        settings = json.loads((root / "read-options.json").read_text())
        request = {
            "data": str(root / "data.csv"), "schema": str(root / manifest["schema_file"]),
            "delimiter": settings["delimiter"], "pandas_options": settings["pandas_options"],
        }
        results = []
        # Fresh process for each replay attempt: no schema/global state carries over.
        for _ in range(manifest["reduction"]["stability_runs"]):
            process = subprocess.Popen(
                [sys.executable, str(root / "_validator.py")], stdin=subprocess.PIPE,
                stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, text=True,
                env={**os.environ, "PYTHONDONTWRITEBYTECODE": "1", "OPENBLAS_NUM_THREADS": "1",
                     "OMP_NUM_THREADS": "1", "POLARS_MAX_THREADS": "2",
                     "FAILUREPACK_PARENT_PID": str(os.getpid())},
                start_new_session=True,
            )
            try:
                stdout, _ = process.communicate(
                    json.dumps(request) + "\n", timeout=manifest["reduction"]["timeout_seconds"],
                )
            finally:
                try:
                    os.killpg(process.pid, signal.SIGKILL)
                except (ProcessLookupError, PermissionError):
                    if process.poll() is None:
                        try:
                            process.kill()
                        except (ProcessLookupError, PermissionError):
                            pass
                process.wait(timeout=2)
                for stream in (process.stdin, process.stdout):
                    if stream is not None:
                        try:
                            stream.close()
                        except OSError:
                            pass
            result = json.loads(stdout)
            if result["status"] != "ok":
                raise ValueError("validator error: " + result.get("message", "unknown"))
            if result.get("parsed_rows") != manifest["result"]["rows"]:
                raise ValueError("parsed row count does not match the bundle manifest")
            results.append(manifest["target"]["id"] in {f["id"] for f in result["failures"]})
        if len(set(results)) != 1:
            raise ValueError("selected failure is nondeterministic")
        print(json.dumps({"reproduced": results[0], "target": manifest["target"]["id"],
                          "rows": manifest["result"]["rows"]}, sort_keys=True))
        return 0 if results[0] else 1
    except (OSError, ValueError, KeyError, TypeError, RecursionError, importlib.metadata.PackageNotFoundError,
            subprocess.TimeoutExpired) as exc:
        print("replay error: " + str(exc), file=sys.stderr)
        return 2


if __name__ == "__main__":
    raise SystemExit(main())
